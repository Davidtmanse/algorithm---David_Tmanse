### 829.模拟队列
```
#include <iostream>

using namespace std;

const int N = 100010;
int queue[N], hh = 0, tt = -1;

void push(int x){
    queue[ ++ tt] = x;

}

void empty(){
    if (hh <= tt) cout << "NO" << endl;
    else cout << "YES" << endl;
}

void pop(){
    hh ++;
}

void query(){

    int num = queue[hh];
    cout << num << endl; 
}


int main(){
    int m;
    cin >> m;
    
    while (m --){
        string op;
        cin >> op;
        
        int x;
        
        if (op == "push"){
            cin >> x;
            push(x);
        }else if (op == "empty"){
            empty();
        }else if (op == "pop"){
            if (hh <= tt) pop();
        }else if (op == "query"){
            if (hh <= tt) query();
        }
    }
    
    return 0;
}
```
