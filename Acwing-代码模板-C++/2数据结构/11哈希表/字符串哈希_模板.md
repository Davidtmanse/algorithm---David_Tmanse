#### 核心思想：将字符串看成P进制数，P的经验值是131或13331，取这两个值的冲突概率低小技巧：取模的数用2^64，这样直接用unsigned long long存储，溢出的结果就是取模的结果
```
typedef unsigned long long ULL;
ULL h[N], p[N]; // h[k]存储字符串前k个字母的哈希值, p[k]存储 P^k mod 2^64

// 初始化
p[0] = 1;
for (int i = 1; i <= n; i ++ )
{
    h[i] = h[i - 1] * P + str[i];
    p[i] = p[i - 1] * P;
}

// 计算子串 str[l ~ r] 的哈希值
ULL get(int l, int r)
{
    return h[r] - h[l - 1] * p[r - l + 1];
}

# 841.字符串哈希
```
#include <iostream>
#include <algorithm>

using namespace std;

typedef unsigned long long ULL;

const int N = 100010;
const int P = 131;      // 经验值：P=131 或 13331，冲突概率极低

int n, m;
char str[N];            // 字符串（从下标 1 开始存）
ULL h[N];               // h[i]：前缀 str[1..i] 的哈希值
ULL p[N];               // p[i]：P^i，用来对齐权重

/*
  取子串 str[l..r] 对应的哈希值
*/
ULL get(int l, int r)
{
    /*
      类比前缀和：sum(l,r) = s[r] - s[l-1]
      但字符串是“P进制数”，高位在左，需要把高位对齐：

      h[r] = str[1]*P^(r-1) + ... + str[l]*P^(r-l) + ... + str[r]
      h[l-1]*P^(r-l+1) = str[1]*P^(r-1) + ... + str[l-1]*P^(r-l+1)

      相减后剩下的正好是：
      str[l]*P^(r-l) + ... + str[r]  → 这就是子串 [l,r] 的哈希
    */
    return h[r] - h[l - 1] * p[r - l + 1];
}

int main()
{
    scanf("%d%d", &n, &m);
    scanf("%s", str + 1);     // 从 1 开始存，方便算前缀

    p[0] = 1;                 // P^0 = 1
    for (int i = 1; i <= n; i ++ )
    {
        // 前缀哈希递推：前面整体升一位 + 新字符
        h[i] = h[i - 1] * P + str[i];
        // 预存 P 的幂次
        p[i] = p[i - 1] * P;
    }

    while (m -- )
    {
        int l1, r1, l2, r2;
        scanf("%d%d%d%d", &l1, &r1, &l2, &r2);

        // O(1) 判断两个子串是否相同
        if (get(l1, r1) == get(l2, r2)) puts("Yes");
        else puts("No");
    }

    return 0;
}
