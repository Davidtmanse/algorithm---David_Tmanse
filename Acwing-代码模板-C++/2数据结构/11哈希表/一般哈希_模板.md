# 840.模拟散列表
### 开放寻址法
```
#include <cstring>
#include <iostream>

using namespace std;

/* 开放寻址法经验值：
   1. N 取质数（减少冲突）
   2. N 要比数据量 n 大 2~3 倍（保证空房多，探测次数少）
*/
const int N = 200003;  

/* 标记“空房间”的特殊值
   0x3f3f3f3f ≈ 1e9+，题目数据一般不到这么大，用来当“没有值”
*/
const int null = 0x3f3f3f3f;

int h[N];  // 哈希表数组（存的是插入的数本身）

/* 找 x 应该待的位置
   返回值有两种情况：
   1. x 已经在表里 → 返回 x 的下标
   2. x 不在表里 → 返回第一个空位的下标（用来插 x）
*/
int find(int x)
{
    // 👇 处理负数取模：C++ 里 -3 % 5 = -3，加 N 再 % N 保证非负
    int t = (x % N + N) % N;

    // 只要没遇到空房，且当前房住的也不是 x 本人 → 继续往后找
    while (h[t] != null && h[t] != x)
    {
        t ++ ;              // 线性探测：往后走一间房
        if (t == N) t = 0;  // 走到数组末尾，循环回开头
    }

    return t;  // 返回找到的位置（要么是空位，要么是 x）
}

int main()
{
    /*
       memset 是按字节赋值的
       0x3f 是一个字节 → int 有 4 个字节 → 0x3f3f3f3f
       所以 h 数组所有元素都变成了 null
    */
    memset(h, 0x3f, sizeof h);

    int n;
    scanf("%d", &n);

    while (n -- )
    {
        char op[2];
        int x;
        scanf("%s%d", op, &x);

        // I x : 插入
        if (*op == 'I')
        {
            int pos = find(x);  // 找到 x 该待的位置
            h[pos] = x;         // 住进去（如果本来就在，重复赋值也没事）
        }
        // Q x : 查询
        else
        {
            int pos = find(x);
            if (h[pos] == null) puts("No");  // 找到的是空房 → 没插过
            else puts("Yes");                // 找到的是 x → 存在
        }
    }

    return 0;
}

### 拉链发
```
#include <cstring>
#include <iostream>

using namespace std;

const int N = 100003;   // 哈希表长度（质数，减少冲突）

int h[N];    // 每个桶的头指针：h[k] = 链表中第一个节点的下标
int e[N];    // 节点的值数组（value）
int ne[N];   // 节点的 next 指针数组
int idx;     // 当前可用节点编号

/* 插入 x */
void insert(int x)
{
    // 1. 算桶号（处理负数取模）
    int k = (x % N + N) % N;

    // 2. 头插法（链表最前面挂新节点）
    e[idx] = x;          // 新节点存值
    ne[idx] = h[k];      // 新节点的 next 指向原来的头
    h[k] = idx;          // 桶头更新成新节点
    idx ++ ;             // 全局节点号 +1
}

/* 查询 x 是否存在 */
bool find(int x)
{
    int k = (x % N + N) % N;

    // 遍历 k 号桶对应的链表
    for (int i = h[k]; i != -1; i = ne[i])
    {
        if (e[i] == x)   // 找到一样的元素
            return true;
    }

    return false;         // 扫完链表没找到
}

int main()
{
    int n;
    scanf("%d", &n);

    // h[] 初始化为 -1，表示“空链表”（和链表头插法约定一致）
    memset(h, -1, sizeof h);

    while (n -- )
    {
        char op[2];
        int x;
        scanf("%s%d", op, &x);

        if (*op == 'I')
        {
            insert(x);
        }
        else
        {
            if (find(x)) puts("Yes");
            else puts("No");
        }
    }

    return 0;
}
