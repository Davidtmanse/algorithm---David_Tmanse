###  154.滑动窗口
```
#include <iostream>

using namespace std;

const int N = 1000010;
int a[N], q[N];

int main(){
    int n, k;
    cin >> n >> k;
    
    for (int i = 0; i < n; i ++) cin >> a[i];
    
    int hh = 0, tt = -1;
    for (int i = 0; i < n; i ++){
        if (hh <= tt && i - k + 1 > q[hh]) hh ++; // 走满队列之后继续往后走，所以才hh ++；
        
        while (hh <= tt && a[q[tt]] >= a[i]) tt --;
        // 始终保持q[0]的值为 该窗口的最小值的下标
        q[ ++ tt ] = i;
        
        if (i >= k - 1) cout << a[q[hh]] << ' ';
    }
    
    puts("");
    
    hh = 0, tt = -1;
    for (int i = 0; i < n ; i ++){
        if (hh <= tt && i - k + 1 > q[hh]) hh ++;
        
        while (hh <= tt && a[q[tt]] <= a[i]) tt --;
        q[ ++ tt ] = i;
        
        if (i >= k - 1) cout << a[q[hh]] << ' ';
    }
    
    puts("");
    
    
    return 0;
}
```
