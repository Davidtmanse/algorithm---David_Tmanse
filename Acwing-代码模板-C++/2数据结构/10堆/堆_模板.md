```
// h[N]存储堆中的值, h[1]是堆顶，x的左儿子是2x, 右儿子是2x + 1
// ph[k]存储第k个插入的点在堆中的位置
// hp[k]存储堆中下标是k的点是第几个插入的
int h[N], ph[N], hp[N], size;

// 交换两个点，及其映射关系
void heap_swap(int a, int b)
{
    swap(ph[hp[a]],ph[hp[b]]);
    swap(hp[a], hp[b]);
    swap(h[a], h[b]);
}

void down(int u)
{
    int t = u;
    if (u * 2 <= size && h[u * 2] < h[t]) t = u * 2;
    if (u * 2 + 1 <= size && h[u * 2 + 1] < h[t]) t = u * 2 + 1;
    if (u != t)
    {
        heap_swap(u, t);
        down(t);
    }
}

void up(int u)
{
    while (u / 2 && h[u] < h[u / 2])
    {
        heap_swap(u, u / 2);
        u >>= 1;
    }
}

// O(n)建堆
for (int i = n / 2; i; i -- ) down(i);
```

# 838.堆排序
```
#include <iostream>
#include <algorithm>

using namespace std;

const int N = 100010;

int n, m;
int h[N], cnt;

void down(int u) // 保证以 u 为根的子树满足“小根堆”性质（父节点 ≤ 子节点）。
{
    int t = u;
    if (u * 2 <= cnt && h[u * 2] < h[t]) t = u * 2;
    if (u * 2 + 1 <= cnt && h[u * 2 + 1] < h[t]) t = u * 2 + 1;
    if (u != t)
    {
        swap(h[u], h[t]);
        down(t);
    }
}

int main()
{
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i ++ ) scanf("%d", &h[i]);
    cnt = n; // 表示当前堆里有多少个元素

    for (int i = n / 2; i; i -- ) down(i);

    while (m -- )
    {
        printf("%d ", h[1]); // 输出最小
        h[1] = h[cnt -- ]; //尾元素顶替根
        down(1); // 恢复堆
    }

    puts("");

    return 0;
}
```
***

给定 `n` 个数，用小根堆排序，输出前 `m` 个最小的数。
数组 `h[N]` 用 `1` 下标 存一棵完全二叉树：
- `h[u]` 的左儿子：`h[u * 2]`
- `h[u]` 的右儿子：`h[u * 2 + 1]`
- 根节点永远是 `h[1]`

# 839.模拟堆
```
#include <iostream>
#include <algorithm>
#include <string.h>

using namespace std;

const int N = 100010;

int h[N];     // 堆数组（1 下标）
int ph[N];    // pointer heap: ph[k] = 第 k 个插入的数在 h[] 的下标
int hp[N];    // heap pointer: hp[i] = h[i] 是第几个插入的数
int cnt;      // 当前堆里有多少元素

/* 交换堆中 a 和 b 两个位置（连带映射一起换） */
void heap_swap(int a, int b)
{
    swap(ph[hp[a]], ph[hp[b]]); // 外部编号对应的下标互换
    swap(hp[a], hp[b]);         // 堆位置对应的外部编号互换
    swap(h[a], h[b]);           // 真正的值互换
}

/* 向下调整（小根堆） */
void down(int u)
{
    int t = u;
    if (u * 2 <= cnt && h[u * 2] < h[t]) t = u * 2;
    if (u * 2 + 1 <= cnt && h[u * 2 + 1] < h[t]) t = u * 2 + 1;
    if (u != t)
    {
        heap_swap(u, t);
        down(t);
    }
}

/* 向上调整（插入 / 改小值时用） */
void up(int u)
{
    while (u / 2 && h[u] < h[u / 2])  // 父节点存在且比父节点小
    {
        heap_swap(u, u / 2);
        u >>= 1;                      // 跳到父节点继续看
    }
}

int main()
{
    int n, m = 0;   // m 记录是第几个插入的（从 1 开始）
    scanf("%d", &n);

    while (n -- )
    {
        char op[5];
        int k, x;
        scanf("%s", op);

        /* I x : 插入 x */
        if (!strcmp(op, "I"))
        {
            scanf("%d", &x);
            cnt ++ ;     // 堆尾加一个位置
            m ++ ;       // 插入序号 +1

            ph[m] = cnt; // 第 m 个插入的点在堆尾
            hp[cnt] = m; // 堆尾这个点就是第 m 个插入的
            h[cnt] = x;

            up(cnt);     // 新元素上浮恢复堆性质
        }
        /* PM : 输出最小值 */
        else if (!strcmp(op, "PM"))
        {
            printf("%d\n", h[1]);
        }
        /* DM : 删除堆顶（最小值） */
        else if (!strcmp(op, "DM"))
        {
            heap_swap(1, cnt); // 堆顶和最后一个换
            cnt -- ;           // 逻辑删除堆尾
            down(1);           // 新堆顶下沉
        }
        /* D k : 删除第 k 个插入的数 */
        else if (!strcmp(op, "D"))
        {
            scanf("%d", &k);
            k = ph[k];         // ⭐ 把“第 k 个插入”转成堆下标

            heap_swap(k, cnt); // 和堆尾交换
            cnt -- ;
            up(k);             // 不知道变大了还是变小了，两个都试
            down(k);
        }
        /* C k x : 把第 k 个插入的数改成 x */
        else
        {
            scanf("%d%d", &k, &x);
            k = ph[k];         // 转成堆下标
            h[k] = x;
            up(k);             // 可能变大或变小
            down(k);
        }
    }

    return 0;
}
```
