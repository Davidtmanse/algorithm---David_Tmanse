#### 常见模型：找出每个数左边离它最近的比它大/小的数
```
int tt = 0;
for (int i = 1; i <= n; i ++){
    while (tt & check(stk[tt], i)) tt --;
    stk[ ++ t] = i;
}
```

### 830.单调栈
```
#include <iostream>

using namespace std;

const int N = 100010;

int stk[N], tt;

int main()
{
    int n;
    cin >> n;
    while (n -- )
    {
        int x;
        scanf("%d", &x);
        while (tt && stk[tt] >= x) tt -- ;
        if (!tt) printf("-1 ");
        else printf("%d ", stk[tt]);
        stk[ ++ tt] = x;
    }

    return 0;
}
```
