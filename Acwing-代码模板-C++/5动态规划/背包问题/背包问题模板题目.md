# 01背包问题
#### 一维优化后的代码
```
#include <iostream>
#include <algorithm>

using namespace std;

const int N = 1010;

int n, m;              // n：物品个数，m：背包容量
int v[N], w[N];        // v[i]：第 i 件物品的体积；w[i]：第 i 件物品的价值
int f[N];              // 一维滚动数组！外层循环进行到第 i 轮时，
                       // f[j] = 「只考虑前 i 件物品、容量为 j 时能获得的最大价值」
                       // 它就是二维写法 f[i][j] 那一行的「压缩版」——
                       // 因为第 i 行只依赖第 i-1 行，算完一行后旧行没用了，直接原地覆盖
                       // 全局数组自动清零，所以 f[0..m] 初始都是 0（边界：没有物品时价值为 0）

int main()
{
    cin >> n >> m;

    for (int i = 1; i <= n; i ++ ) cin >> v[i] >> w[i];

    // 外层循环 i：一件一件地把物品加进来
    // 每处理完一件，整个 f[] 就由「前 i-1 件」的答案原地滚动成「前 i 件」的答案
    for (int i = 1; i <= n; i ++ )
        // 内层循环 j 必须从大到小（倒序），这是整个一维优化最关键的一行！
        // 原因：转移要用到 f[j - v[i]]，它必须是「上一轮（前 i-1 件）」的旧值
        //       倒序时 j 大的先算，j - v[i] 更小、还没被本轮更新，读到的自然是旧值
        //       若改成正序，小的 j 先被更新成「已经拿过第 i 件」的新值，
        //       大的 j 再拿它去转移，第 i 件就被重复拿了（退化成完全背包，答案偏大）
        // j 从 m 开始、到 v[i] 为止：j < v[i] 时连第 i 件都装不下，不可能是最优，直接跳过
        for (int j = m; j >= v[i]; j -- )
            // 与二维版本的转移完全等价：
            //   不选第 i 件 -> 保持 f[j] 原样（继承上一轮的旧值，什么都不做，天然成立）
            //   选第 i 件   -> 腾出 v[i] 的容量给前 i-1 件：f[j - v[i]] + w[i]
            //   两者取较大者
            f[j] = max(f[j], f[j - v[i]] + w[i]);

    // 处理完全部 n 件物品后，f[m] 就是容量 m 能装下的最大价值
    cout << f[m] << endl;

    return 0;

```

#### 二维写法
```
#include <iostream>
#include <algorithm>

using namespace std;

const int N = 1010;          // 数组多开一点空间，防止越界

int n, m;                    // n：物品个数，m：背包容量
int v[N], w[N];              // v[i]：第 i 件物品的体积；w[i]：第 i 件物品的价值
int f[N][N];                 // f[i][j]：只考虑前 i 件物品、容量为 j 时能获得的最大价值
                             // 全局数组自动清零，所以 f[0][j] = 0（边界：没有物品可选）

int main() {
    cin >> n >> m;

    // 读入每件物品的体积和价值（下标从 1 开始，方便和物品编号对应）
    for (int i = 1; i <= n; i ++) cin >> v[i] >> w[i];

    // 自底向上：先只考虑 1 件，再考虑 2 件……直到考虑完全部 n 件
    for (int i = 1; i <= n; i ++)
        for (int j = 0; j <= m; j ++) {
            // ---------- 把「前 i 件、容量 j 的所有选法」这个集合切成两半 ----------

            // 子集 A：不选第 i 件
            // 等价于「在前 i-1 件里选、容量仍为 j」，最大值就是上一行同列的 f[i-1][j]，
            // 无条件直接继承（不选永远合法，不需要腾空间）
            f[i][j] = f[i - 1][j];

            // 子集 B：选第 i 件（前提：容量 j 至少放得下 v[i]）
            // 强制拿走第 i 件，剩下 j - v[i] 的容量交给前 i-1 件去装，
            // 最大值 = f[i-1][j-v[i]] + w[i]；和子集 A 取较大者
            if (j >= v[i])
                f[i][j] = max(f[i][j], f[i - 1][j - v[i]] + w[i]);
        }

    // 考虑完全部 n 件物品、容量正好为 m 时，就是最终答案
    cout << f[n][m] << endl;

    return 0;
}
```

# 完全背包问题
#### 优化前的暴力写法
```
#include <iostream>
#include <algorithm>

using namespace std;

const int N = 1010;

int n, m;              // n：物品种数，m：背包容量
int v[N], w[N];        // v[i]：第 i 种物品的体积；w[i]：第 i 种物品的价值
int f[N][N];           // f[i][j]：只考虑前 i 种物品、容量为 j 时能获得的最大价值
                       // 完全背包 vs 01背包：每种物品可以拿任意多件（0 件、1 件、2 件……）
                       // 全局数组自动清零 → f[0][j] = 0（一种物品都不考虑，价值为 0，递推起点）

int main()
{
    cin >> n >> m;

    for (int i = 1; i <= n; i ++ ) cin >> v[i] >> w[i];

    // 外层循环 i：逐种考虑物品（一种一种加入决策）
    for (int i = 1; i <= n; i ++ )
        // 中层循环 j：枚举容量，从小到大（二维版本容量顺序无所谓）
        for (int j = 0; j <= m; j ++ )
            // 内层循环 k：第 i 种物品到底拿几件？—— 这是完全背包的灵魂
            //
            // 把「前 i 种、容量 j 的所有选法」这个集合，按「第 i 种拿了 k 件」
            // 切成互不相交的子集：k = 0, 1, 2, ..., 直到 k*v[i] > j（装不下就停）
            //   子集 k 的含义：强制拿 k 件第 i 种物品，剩下的容量 j - k*v[i]
            //                  全部交给「前 i-1 种物品」去装（这是已经算好的子问题）
            //   子集 k 的最大值 = f[i-1][j - k*v[i]] + k*w[i]
            //                   （前 i-1 种装剩下的容量 + 这 k 件的价值）
            //
            // 所有子集互不重叠、并起来正好是全集
            // → 全集的 max = 所有子集 max 里的 max
            // → 用 f[i][j] 做累积器，k 每取一个值就比一次
            //   注意 k=0 时就是 f[i-1][j]（一件不拿），所以「不选」天然被覆盖，
            //   不需要像 01 背包那样单独写一行继承
            for (int k = 0; k * v[i] <= j; k ++ )
                f[i][j] = max(f[i][j], f[i - 1][j - v[i] * k] + w[i] * k);

    // 所有 n 种物品都考虑完、容量正好 m 时，就是最终答案
    cout << f[n][m] << endl;

    return 0;
}
```

#### 第一次优化_二维
```
#include <iostream>
#include <algorithm>

using namespace std;

const int N = 1010;

int n, m;              // n：物品种数，m：背包容量
int v[N], w[N];        // v[i]：第 i 种物品的体积；w[i]：第 i 种物品的价值
int f[N][N];           // f[i][j]：只考虑前 i 种物品、容量为 j 时能获得的最大价值
                       // 完全背包：每种物品可拿任意多件
                       // 全局数组自动清零 → f[0][j] = 0（没有物品可选，递推起点）

int main()
{
    cin >> n >> m;
    for (int i = 1; i <= n; i ++ ) cin >> v[i] >> w[i];

    // 外层循环 i：逐种考虑物品（一种一种加入决策）
    for (int i = 1; i <= n; i ++ )
        // 中层循环 j：枚举容量
        for (int j = 1; j <= m; j ++ ) {
            // 情况 1：容量 j 连一件第 i 种都装不下
            //   → 第 i 种一件都拿不了，答案只能来自「前 i-1 种」：直接继承上一行
            if (j < v[i])
                f[i][j] = f[i - 1][j];
            // 情况 2：容量装得下第 i 种 → 把集合切成两个子集，取 max：
            else {
                // ① 不拿第 i 种：答案 = f[i-1][j]（前 i-1 种、容量 j）
                //
                // ② 至少拿 1 件第 i 种：花掉 v[i] 的容量，剩下 j-v[i] 的容量
                //    ——关键区别在这里——
                //    剩余容量面对的还是「前 i 种」物品（因为还能再拿第 i 种！）
                //    所以要用「本行」的 f[i][j-v[i]]，而不是上一行的 f[i-1][j-v[i]]
                //    f[i][j-v[i]] 内部已经包含「再拿 0 件、1 件、2 件……第 i 种」
                //    的所有可能（算它时同样用本行转移，递归地连锁下去）
                //    → 这等价于上一课三重循环里枚举 k = 1,2,3,... 的所有情况
                //    所以一次 max 就覆盖了所有 k，第三重循环被彻底消灭
                f[i][j] = max(f[i - 1][j], f[i][j - v[i]] + w[i]);
            }
        }

    // 所有 n 种都考虑完、容量正好 m → 最终答案
    cout << f[n][m] << endl;

    return 0;
}
```

#### 最终优化_一维
```
#include <iostream>
#include <algorithm>

using namespace std;

const int N = 1010;

int n, m;              // n：物品种数，m：背包容量
int v[N], w[N];        // v[i]：第 i 种物品的体积；w[i]：第 i 种物品的价值
int f[N];              // 一维滚动数组：f[j] = 当前已考虑的物品范围下、容量 j 能装出的最大价值
                       // 全局数组自动清零 → 初始全部为 0（一件物品都没考虑，价值为 0）

int main()
{
    cin >> n >> m;

    // 读入每种物品的体积和价值（下标从 1 开始，方便和物品编号对应）
    for (int i = 1; i <= n; i ++ ) cin >> v[i] >> w[i];

    // 外层循环 i：一种一种地把物品加入考虑范围
    // 每轮结束时，f[] 就从「前 i-1 种」的答案滚动成了「前 i 种」的答案
    for (int i = 1; i <= n; i ++ )
        // 内层循环 j：容量从小到大【正序】枚举 —— 完全背包一维版的关键！
        //
        // 转移方程：f[j] = max(f[j], f[j - v[i]] + w[i])
        //   · 不拿第 i 种：f[j] 保持原样（等号左边 f[j] 自己就是旧值）
        //   · 拿第 i 种：花掉 v[i] 容量，剩下 j - v[i] 的容量仍然交给
        //     「前 i 种」继续装（注意是"前 i 种"不是"前 i-1 种"！），
        //     价值 = f[j - v[i]] + w[i]
        //
        // 为什么 j 必须正序？（对照上一课的二维优化版）
        //   完全背包允许第 i 种拿多件 → 转移时读的 f[j - v[i]] 必须是
        //   「本轮已经更新过」的新值（= 二维优化版的 f[i][j - v[i]]，本行），
        //   它内部已经包含「再拿 1 件、再拿 2 件……」的全部可能 → 支持反复拿。
        //   正序时 j 小的格子先被更新，轮到 j 时，j - v[i] < j 早已更新完，正好满足。
        //
        //   对比 01 背包一维版：j 必须【倒序】，读的才是上一轮的旧值
        //   f[i-1][j - v[i]]（上一行），防止同一件被拿两次。
        //   —— 循环方向不同，读到的"行"就不同，语义就完全不同：
        //      正序 = 完全背包（可重复拿）；倒序 = 01 背包（每件最多一次）
        //
        //   j 从 v[i] 开始：j < v[i] 时装不下第 i 种，不可能是转移来源，跳过即可
        for (int j = v[i]; j <= m; j ++ )
            // f[j - v[i]] 在正序下是本轮新值 → 第 i 种可以拿 0 件、1 件、2 件……
            // 取「不拿」与「拿」中的较大者，滚动更新
            f[j] = max(f[j], f[j - v[i]] + w[i]);

    // 考虑完全部 n 种物品、容量正好为 m 时，就是最终答案
    cout << f[m] << endl;

    return 0;
}
```

# 多重背包问题
```
#include <iostream>
#include <algorithm>

using namespace std;

const int N = 110;

int n, m;              // n：物品种数，m：背包容量
int v[N], w[N], s[N];  // v[i]/w[i]：第 i 种物品的体积/价值；s[i]：第 i 种最多有 s[i] 件
int f[N][N];           // f[i][j]：只考虑前 i 种物品、容量为 j 时能获得的最大价值
                       // 全局数组自动清零 → f[0][j] = 0（一件不选，递推起点）

int main()
{
    cin >> n >> m;

    // 每种物品三个参数：体积、价值、件数上限
    for (int i = 1; i <= n; i ++ ) cin >> v[i] >> w[i] >> s[i];

    // 外层循环 i：一种一种地把物品加入考虑范围
    for (int i = 1; i <= n; i ++ )
        // 中层循环 j：枚举容量 0..m（二维写法容量顺序无所谓）
        for (int j = 0; j <= m; j ++ )
            // 内层循环 k：第 i 种物品到底拿几件？—— 多重背包的灵魂
            //
            // 集合划分：把「前 i 种、容量 j 的所有选法」按「第 i 种拿了 k 件」
            // 切成互不相交的子集，全集的 max = 各子集 max 的 max。
            // 与完全背包共用同一条转移方程，唯一的区别是 k 多了一个上限：
            //   01 背包：  k = 0 … 1                    （每件只有 1 个）
            //   完全背包：k = 0 … ⌊j/v[i]⌋              （件数无限，只被容量卡）
            //   多重背包：k = 0 … min(s[i], ⌊j/v[i]⌋)   （件数有 s[i] 个上限）
            //
            // 循环条件是两个上限同时生效，缺一不可：
            //   · k <= s[i]   ：件数上限——超过它，这种物品就没有库存了
            //   · k*v[i] <= j ：容量上限——超过它就装不下；
            //                   且若去掉，j - v[i]*k 会算出负数下标，数组越界
            // 只写前者 → 越界；只写后者 → 退化成完全背包
            //
            // 子集 k 的最大值 = f[i-1][j - k*v[i]] + k*w[i]
            //   （强制拿 k 件第 i 种，剩下 j - k·v[i] 的容量全部交给
            //     前 i-1 种去装——那是上一行已经算好的子问题）
            // 注意 k = 0 时就是 f[i-1][j]（一件不拿），
            // 所以「不选」天然被覆盖，不需要单独写一行继承；
            // f[i][j] 在这里当累积器，k 每取一个值就比较一次、留下最大
            for (int k = 0; k <= s[i] && k * v[i] <= j; k ++ )
                f[i][j] = max(f[i][j], f[i - 1][j - v[i] * k] + w[i] * k);

    // 考虑完全部 n 种物品、容量正好为 m 时，就是最终答案
    cout << f[n][m] << endl;
    return 0;
}
```

# 多重背包问题2
```
#include <iostream>
#include <algorithm>

using namespace std;

// 多重背包 · 二进制拆分优化版
//
// 朴素三重循环（上一课）复杂度 O(n·m·s)，当 s 很大（如 2000）时会超时。
// 优化思路：把「第 i 种共 s 件」打包成 1, 2, 4, 8, ... （2 的幂）+ 一个剩余捆，
//           每一捆当成「一件」01 背包物品，从而把多重背包转化为 01 背包。
//
// 为什么按 2 的幂打包就够？
//   1, 2, 4, ..., 2^(k-1) 能凑出 0 ~ 2^k - 1 的所有整数（2 的幂的经典性质）；
//   再补一个「剩余捆 r」（r < 2^k），就能凑出 0 ~ s 的所有件数。
//   于是「拿 k 件第 i 种」的任何决策，都等价于「这 log(s) 捆各自选或不选」。
//   件数上限 s 也被天然保证：捆的总件数正好是 s，全选也只能拿 s 件，不会多拿。
//
// 规模估算：n = 1000 种、s ≤ 2000 时，每种最多拆出 ⌊log₂2000⌋ + 1 ≈ 12 捆，
//           总物品数 ≤ 1000 × 12 = 12000 件，所以 N = 12010。

const int N = 12010, M = 2010;   // N：拆分后物品总数上限；M：背包容量上限（f 只需开到容量大小）

int n, m;                        // n：原始物品种数，m：背包容量
int v[N], w[N];                  // 拆分后每「捆」的总体积、总价值（捆内物品绑定，视作一件）
int f[M];                       // 一维 01 背包数组（就是第二课的滚动数组模板）

int main()
{
    cin >> n >> m;

    // ---------- 第一阶段：读入 + 二进制拆分 ----------
    int cnt = 0;                          // cnt：拆分后的捆计数器（新的「物品」总数）
    for (int i = 1; i <= n; i ++ )
    {
        int a, b, s;                      // a：单件体积，b：单件价值，s：该种最多拿 s 件
        cin >> a >> b >> s;

        // 按 1, 2, 4, 8, ... 依次打包：
        //   k 是当前捆的件数；只要还剩的 s 够（k <= s），就打包一捆 k 件
        //   一捆 k 件 = 一件体积 a*k、价值 b*k 的 01 物品
        int k = 1;
        while (k <= s)
        {
            cnt ++ ;
            v[cnt] = a * k;               // 这捆的总体积
            w[cnt] = b * k;               // 这捆的总价值
            s -= k;                       // 剩余件数减掉已打包的 k 件
            k *= 2;                       // 下一捆件数翻倍：1 → 2 → 4 → 8 → ...
        }

        // 2 的幂打包完后若还有剩余（0 < s < k），剩余的 s 件单独打成一捆。
        // 例：s = 13 → 拆出 1, 2, 4（此时 s 剩 6，k = 8 装不下）→ 再补一捆 6 件
        //     1+2+4 能凑 0~7，加上捆 6 后能凑 0~13，所有件数全覆盖。
        if (s > 0)
        {
            cnt ++ ;
            v[cnt] = a * s;
            w[cnt] = b * s;
        }
    }

    n = cnt;                              // 从现在起，问题就是纯粹的 01 背包：n 件（捆）物品

    // ---------- 第二阶段：标准 01 背包 · 一维倒序（第二课的模板原样搬来） ----------
    // 注意捆内物品是绑定的（要么整捆拿、要么不拿），且每捆只有一件 → 01 背包
    // j 必须倒序：保证 f[j - v[i]] 读到的是上一轮旧值，同一捆不会被拿两次
    for (int i = 1; i <= n; i ++ )
        for (int j = m; j >= v[i]; j -- )
            f[j] = max(f[j], f[j - v[i]] + w[i]);

    cout << f[m] << endl;

    return 0;
}
```

# 分组背包问题
```
#include <iostream>
#include <algorithm>

using namespace std;

// 分组背包：n 组物品，每组 s[i] 件，约束是「每组至多挑一件」（可以一件都不挑）。
//
// 与前几课的对比：
//   01 背包 / 多重背包：问「第 i 种拿几件」→ 集合按「件数 k」划分；
//   分组背包：        问「第 i 组挑哪一件」→ 集合按「选组内哪一件」划分：
//       子集「不选」       → f[j]（保持上一组的结果，不用写，max 天然带上）
//       子集「选第 k 件」  → f[j - v[i][k]] + w[i][k]（要求 v[i][k] <= j）
//   每个子集互不重叠、并起来是全集 → 全集 max = 逐个比较取 max。
//
// 复杂度 O(m · Σs[i])，即所有物品件数 × 容量。

const int N = 110;

int n, m;                       // n：组数，m：背包容量
int v[N][N], w[N][N];           // v[i][k]、w[i][k]：第 i 组第 k 件物品的体积、价值
int s[N];                       // s[i]：第 i 组的物品件数
int f[N];                       // f[j]：处理完当前组后，容量 j 能获得的最大价值
                                // （一维滚动数组，第二课 01 背包同款）
                                // 全局数组自动清零 → 还没处理任何组时 f[j] = 0，是递推起点

int main()
{
    cin >> n >> m;

    // 读入：先读第 i 组的件数 s[i]，再读这一组每件的体积和价值
    for (int i = 1; i <= n; i ++ )
    {
        cin >> s[i];
        for (int j = 0; j < s[i]; j ++ )        // 组内下标从 0 开始
            cin >> v[i][j] >> w[i][j];
    }

    // 外层循环 i：一组一组地考虑（处理完第 i 组，f[] 就表示「前 i 组」的答案）
    for (int i = 1; i <= n; i ++ )

        // 中层循环 j：容量从 m 到 0 倒序枚举（和 01 背包一维倒序同一个原因）
        //
        // 倒序的原因：更新 f[j] 时要读的 f[j - v[i][k]]（下标更小）必须是
        // 「上一组」的旧值。倒序让小的下标还没被本组更新，读到的是旧值
        // → 同一组的两件不可能连环带入（保证每组至多一件）。
        for (int j = m; j >= 0; j -- )

            // 内层循环 k：枚举第 i 组的每一件，看「挑这一件」能否更优
            //
            // 【注意循环嵌套顺序：j 在外、k 在内，不能颠倒！】
            //   正确（本代码）：对每个 j，把组内所有件横向比一遍，
            //                  每件最多给 f[j] 贡献一次更新 → 每组至多带入一件 ✓
            //   错误（k 在外、j 在内）：等价于把组内每件当成独立 01 物品
            //                  各跑一遍背包 → 同组两件可能同时被选上 ✗
            //
            // 「不选第 i 组」这个子集不需要单独写：
            //   k 循环开始前 f[j] 里躺着的就是上一组的结果，
            //   max(f[j], ...) 的第一个参数天然代表了「这组一件不挑」。
            for (int k = 0; k < s[i]; k ++ )

                // 前提：容量 j 至少装得下第 i 组第 k 件，才存在「挑它」这个选项
                // （装不下时 f[j] 保持原值 = 自动回到「不选本组」）
                if (v[i][k] <= j)
                    f[j] = max(f[j], f[j - v[i][k]] + w[i][k]);

    // n 组全部考虑完、容量正好 m 时的最大价值
    cout << f[m] << endl;

    return 0;
}
```
