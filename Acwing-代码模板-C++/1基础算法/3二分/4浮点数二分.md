##### 二分模板题 - C++

```
#include <iostream>

using namespace std;


int main()
{
    double n;
    cin >> n;

    double l = -10000 , r = 10000;
    while (r - l > 1e-8)
    {
        double mid = (l + r)/ 2;
        if (mid * mid * mid >= n) r = mid;
        else l = mid;
    }


    printf("%.6lf" , l);
    return 0;
}
```

### Java实现版本
```
import java.util.*;

public class Main{
    private static final double eps = 1e-8;
    
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        
        double x = sc.nextDouble();
        
        double l = -10000;
        double r = 10000;
        
        while (r - l > eps){
            double mid = (r + l) / 2;
            if (mid * mid * mid >= x) r = mid;
            else l = mid;
        }
        
        System.out.printf("%.6f\n",l);
    }
}
```
