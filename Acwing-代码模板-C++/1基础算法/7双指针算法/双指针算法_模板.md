### 799.[最长连续不重复子序列]()
```
#include <iostream>

using namespace std;

const int N = 100010;
int n;
int a[N] , s[N];

int main()
{
    cin >> n;
    for (int i = 0 ; i < n ; i ++) cin >> a[i];

    int res = 0;
    for (int i = 0 , j = 0; i < n ; i ++)
    {
        s[a[i]] ++;
        while (j < i && s[a[i]] > 1) s[a[j ++]] --;
        res = max(res , i - j + 1);
    }

    cout << res << endl;
    return 0;
}
```
一个小知识点:`s[q[j ++]] --`
等价于：
```
while (j < i && s[q[i]] > 1) {
    int val = q[j];   // 当前左端点的值
    s[val]--;         // 计数减掉
    j++;              // 左指针右移
}

### 800.[数组元素的目标和登录]()
```
#include <iostream>

using namespace std;

const int N = 100010;
int q[N], s[N];

int main(){
    int n, m, x;
    cin >> n >> m >> x;
    
    for (int i = 0; i < n; i ++) cin >> q[i];
    for (int i = 0; i < m; i ++) cin >> s[i];
    
    for (int i = 0, j = m - 1; i < n ; i ++){
        while (j >= 0 && q[i] + s[j] > x) j --;
        if (j >= 0 && q[i] + s[j] == x) cout << i << ' ' << j << endl;
    }
    
    return 0;
}
```

### 2816.[判断子序列]()
自己的写法
```

#include <iostream>

using namespace std;

const int N = 100010;
int a[N], b[N],s[N];

int main(){
    int n, m;
    cin >> n >> m;
    
    for (int i = 0; i < n; i ++) cin >> a[i];
    for (int j = 0; j < m; j ++) cin >> b[j];
    
    for (int i = 0, j = 0; i < n ;){
        while (j < m){
            if (a[i] == b[j]){
                i ++;
                if (i == n){
                    cout << "Yes" << endl;
                    return 0;
                }
            }
            j ++;
        }
        
        if (i == n){
            cout << "Yes" << endl;
            return 0;
        }else {
            cout << "No" << endl;
            return 0;
        }    
    }    
}
```

```
#include <iostream>
#include <cstring>

using namespace std;

const int N = 100010;

int n, m;
int a[N], b[N];

int main()
{
    scanf("%d%d", &n, &m);
    for (int i = 0; i < n; i ++ ) scanf("%d", &a[i]);
    for (int i = 0; i < m; i ++ ) scanf("%d", &b[i]);

    int i = 0, j = 0;
    while (i < n && j < m)
    {
        if (a[i] == b[j]) i ++ ;
        j ++ ;
    }

    if (i == n) puts("Yes");
    else puts("No");

    return 0;
}
