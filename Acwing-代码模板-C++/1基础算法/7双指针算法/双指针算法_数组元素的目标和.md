```
#include <iostream>

using namespace std;

const int N = 100010;
int q[N], s[N];

int main(){
    int n, m, x;
    cin >> n >> m >> x;
    
    for (int i = 0; i < n; i ++) cin >> q[i];
    for (int i = 0; i < m; i ++) cin >> s[i];
    
    for (int i = 0, j = m - 1; i < n ; i ++){
        while (j >= 0 && q[i] + s[j] > x) j --;
        if (j >= 0 && q[i] + s[j] == x) cout << i << ' ' << j << endl;
    }
    
    return 0;
}

### Java写法
```
import java.util.*;

public class Main{
    private static int N = 100010;
    private static int[] a = new int[N];
    private static int[] b = new int[N];
    
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        
        int n = sc.nextInt();
        int m = sc.nextInt();
        int x = sc.nextInt();
        
        for (int i = 0; i < n; i ++) a[i] = sc.nextInt();
        for (int j = 0; j < m; j ++) b[j] = sc.nextInt();
        
        for (int i = 0, j = m - 1; i < n; i ++){
            while (j > 0 && a[i] + b[j] > x) j --;
            if (a[i] + b[j] == x) System.out.println(i + " " + j);
        }
    }
}
