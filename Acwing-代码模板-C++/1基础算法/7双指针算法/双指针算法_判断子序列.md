### 第一种写法
```
#include <iostream>

using namespace std;

const int N = 100010;
int a[N], b[N],s[N];

int main(){
    int n, m;
    cin >> n >> m;
    
    for (int i = 0; i < n; i ++) cin >> a[i];
    for (int j = 0; j < m; j ++) cin >> b[j];
    
    for (int i = 0, j = 0; i < n ;){
        while (j < m){
            if (a[i] == b[j]){
                i ++;
                if (i == n){
                    cout << "Yes" << endl;
                    return 0;
                }
            }
            j ++;
        }
        
        if (i == n){
            cout << "Yes" << endl;
            return 0;
        }else {
            cout << "No" << endl;
            return 0;
        }    
    }    
}
```

### 第二种写法
```
#include <iostream>
#include <cstring>

using namespace std;

const int N = 100010;

int n, m;
int a[N], b[N];

int main()
{
    scanf("%d%d", &n, &m);
    for (int i = 0; i < n; i ++ ) scanf("%d", &a[i]);
    for (int i = 0; i < m; i ++ ) scanf("%d", &b[i]);

    int i = 0, j = 0;
    while (i < n && j < m)
    {
        if (a[i] == b[j]) i ++ ;
        j ++ ;
    }

    if (i == n) puts("Yes");
    else puts("No");

    return 0;
}
```

### Java写法
```
import java.util.*;

public class Main{
    private static int N = 100010;
    private static int[] a = new int[N];
    private static int[] b = new int[N];
    
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        
        int n = sc.nextInt();
        int m = sc.nextInt();
        
        for (int i = 0; i < n; i ++) a[i] = sc.nextInt();
        for (int i = 0; i < m; i ++) b[i] = sc.nextInt();
        
        
        int i, j;
        for (i = 0, j = 0; i < m && j < n; i ++){
            if (a[j] == b[i]) j ++; 
        }
        
        if (j == n) System.out.println("Yes");
        else System.out.println("No");
    }
}
```
